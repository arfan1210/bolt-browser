# Mirror the APK of every published release to the PUBLIC releases repo.
#
# Why: the source repo is private; the public repo (bolt-browser-releases)
# lets anyone download the APK without a GitHub account. ONLY the .apk is
# mirrored — never the source.zip (it contains the signing keystore).
#
# SETUP (one-time, ~30 seconds): this workflow needs a token that can write to
# arfan1210/bolt-browser-releases. The private repo's built-in GITHUB_TOKEN
# cannot cross repos. Add it here:
#   Settings → Secrets and variables → Actions → "New repository secret"
#     Name:  RELEASE_MIRROR_TOKEN
#     Value: a PAT — easiest: classic token with the `repo` scope
#            (fine-grained: Contents: Read+write on bolt-browser-releases)
# Until the secret exists, this workflow exits gracefully (a clear note is
# logged) — releases still land in the private repo normally.
#
# THE RACE (learned on v1.3.80, 2026-11-09): `release: published` fires the
# moment the release OBJECT is created — BEFORE its assets finish uploading.
# 65 releases mirrored fine because runner pickup was slow; v1.3.80's runner
# picked the job up in 2 s, found zero assets ("no .apk asset in source
# release" → exit 26) and the public mirror died while the APK was still
# uploading. The mirror now WAITS for the .apk asset (15 s × 24 = 6 min)
# instead of assuming it exists.
#
# RE-MIRROR ANY TAG: Actions → mirror-release-apk → Run workflow → tag input
# (empty = latest release). No new release needed — this is the recovery path
# for a failed mirror.

name: mirror-release-apk

on:
  release:
    types: [published]
  workflow_dispatch:
    inputs:
      tag:
        description: 'Tag to mirror (e.g. v1.3.80). Empty = latest release.'
        required: false
        default: ''

permissions:
  contents: read

jobs:
  mirror:
    runs-on: ubuntu-latest
    env:
      SRC_REPO: arfan1210/bolt-light-browser
      DST_REPO: arfan1210/bolt-browser-releases
      MIRROR_TOKEN: ${{ secrets.RELEASE_MIRROR_TOKEN }}
    steps:
      - name: Check mirror token
        run: |
          if [ -z "$MIRROR_TOKEN" ]; then
            echo "RELEASE_MIRROR_TOKEN secret is not set — skipping the public mirror."
            echo "Private release is complete. To also publish the APK to the public"
            echo "repo automatically: Settings → Secrets and variables → Actions →"
            echo "New repository secret RELEASE_MIRROR_TOKEN (classic PAT, 'repo' scope)."
          fi

      - name: Mirror APK to public releases repo
        if: env.MIRROR_TOKEN != ''
        env:
          INPUT_TAG: ${{ github.event.inputs.tag }}
        run: |
          set -e
          SRC="https://api.github.com/repos/$SRC_REPO"
          DST="https://api.github.com/repos/$DST_REPO"
          SA="Authorization: Bearer ${{ secrets.GITHUB_TOKEN }}"
          DA="Authorization: Bearer $MIRROR_TOKEN"
          ACC="Accept: application/vnd.github+json"

          # 0. resolve the tag: dispatch input → release event ref → latest release
          TAG="$INPUT_TAG"
          if [ -z "$TAG" ] && [ "$GITHUB_EVENT_NAME" = "release" ]; then TAG="${GITHUB_REF_NAME}"; fi
          if [ -z "$TAG" ]; then
            TAG=$(curl -s -H "$SA" -H "$ACC" "$SRC/releases/latest" \
              | python3 -c "import json,sys;print(json.load(sys.stdin)['tag_name'])")
          fi
          echo "mirroring $TAG ..."

          # 1. public repo must exist (create if the token allows; it is public)
          CODE=$(curl -s -o /dev/null -w "%{http_code}" -H "$DA" -H "$ACC" "$DST")
          if [ "$CODE" != "200" ]; then
            echo "public repo $DST_REPO missing (HTTP $CODE) — creating"
            curl -s -X POST -H "$DA" -H "$ACC" https://api.github.com/user/repos \
              -d '{"name":"bolt-browser-releases","description":"Official releases of Bolt Light Browser — a tiny, fast, privacy-first Android browser. Download the APK here, no login required.","private":false}' \
              | python3 -c "import json,sys;d=json.load(sys.stdin);print(' created:',d.get('html_url') or d.get('message'))"
          fi

          # 2. WAIT for the source release's .apk asset. The published event
          # fires BEFORE assets finish uploading (the v1.3.80 race) — poll
          # 15 s × 24 = 6 min instead of assuming the asset exists.
          ASSET_ID=""
          for i in $(seq 1 24); do
            curl -s -H "$SA" -H "$ACC" "$SRC/releases/tags/$TAG" > /tmp/src.json
            ASSET_ID=$(python3 - << 'PYID'
          import json
          try:
              r = json.load(open('/tmp/src.json'))
              apk = next((a for a in r.get('assets', []) if a['name'].endswith('.apk')), None)
              print(apk['id'] if apk else '')
          except Exception:
              print('')
          PYID
          )
            if [ -n "$ASSET_ID" ]; then break; fi
            echo "no .apk asset on $TAG yet (upload may still be in flight) — retry in 15 s ($i/24)"
            sleep 15
          done
          if [ -z "$ASSET_ID" ]; then
            echo "::error::no .apk asset appeared on $TAG after 6 minutes — nothing to mirror"
            exit 1
          fi

          . <(python3 - << 'PYEOF'
          import json
          r = json.load(open('/tmp/src.json'))
          apk = next((a for a in r['assets'] if a['name'].endswith('.apk')), None)
          assert apk, 'no .apk asset in source release'
          print('ASSET_ID=%d' % apk['id'])
          print('ASSET_NAME=%s' % json.dumps(apk['name']))
          print('REL_NAME=%s' % json.dumps(r['name'] or r['tag_name']))
          payload = {'tag_name': r['tag_name'], 'name': r['name'] or r['tag_name'],
                     'body': r.get('body') or ''}
          open('/tmp/create.json','w').write(json.dumps(payload))
          PYEOF
          )
          echo "asset: $ASSET_NAME (upload-race safe)"

          # 3. download the apk from the private repo
          curl -sL -H "$SA" -H "Accept: application/octet-stream" \
            "$SRC/releases/assets/$ASSET_ID" -o /tmp/mirror.apk
          ls -la /tmp/mirror.apk

          # 4. public release (reuse if it exists)
          curl -s -H "$DA" -H "$ACC" "$DST/releases/tags/$TAG" > /tmp/dst.json
          if python3 -c "import json,sys;sys.exit(0 if 'id' in json.load(open('/tmp/dst.json')) else 1)"; then
            echo "public release already exists — refreshing asset"
          else
            curl -s -X POST -H "$DA" -H "$ACC" "$DST/releases" \
              --data @/tmp/create.json > /tmp/dst.json
            python3 -c "import json;d=json.load(open('/tmp/dst.json'));assert 'id' in d, d.get('message')"
          fi
          for AID in $(python3 -c "import json;print(' '.join(str(a['id']) for a in json.load(open('/tmp/dst.json')).get('assets',[]) if a['name']=='$ASSET_NAME'))"); do
            curl -s -X DELETE -H "$DA" "$DST/releases/assets/$AID" || true
          done

          # 5. upload
          UP=$(python3 -c "import json;print(json.load(open('/tmp/dst.json'))['upload_url'].replace('{?name,label}',''))")
          curl -s -X POST -H "$DA" -H "Content-Type: application/octet-stream" \
            --data-binary "@/tmp/mirror.apk" "$UP?name=$ASSET_NAME" \
            | python3 -c "import json,sys;d=json.load(sys.stdin);print(' uploaded:',d.get('name'),d.get('size'),'bytes  url:',d.get('browser_download_url'))"

          echo "PUBLIC MIRROR DONE: https://github.com/$DST_REPO/releases/tag/$TAG"
